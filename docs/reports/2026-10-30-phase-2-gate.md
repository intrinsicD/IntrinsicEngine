## Method and scripts

The scripts are local evidence (`tasks/evidence/` is not tracked); the three that decide the numbers
are reproduced here.
- **Counting rules:** `outlier_footprint.py`, `pilot_footprint.py` and
  `tasks/evidence/RUNTIME-353/dw_footprint2.py`.
  - A file counts when it names the operation, outside `src/geometry`, `src/graphics`, hand-written
    documentation and the consolidation module's unrelated `Density` member.
  - Own files count whole.
  - In shared files, every definition whose header names the operation counts whole, and so do
    the panel's registration statement and (density weights) the codec namespace block. Every other
    line naming the operation counts once.
  - Tests are apart.
- **`footprint_lines.py`:** applies these rules as line sets, asserts the scripts' totals at
  `f96a13e2d` (423, 429, 883, 1,593), `69ced3262` (1,475) and `b593c0d9e` (1,484), and removes the
  overlaps.
- **Run as:** `python3 framework_reconcile.py 0348dbbed` and `python3 phase2_projection.py 0348dbbed`
  from the repository root. The end revision and the table are read at `0348dbbed`, so later commits do
  not change these results.

```python
# footprint_lines.py
import re, subprocess
from collections import defaultdict
SCOPE = ['src', 'docs/api/generated', 'tests/data/agent_declared_tool_schemas.json']


def read(rev, p):
    return subprocess.run(['git', 'show', f'{rev}:{p}'], capture_output=True, text=True).stdout


def files(rev, pat):
    out = subprocess.run(['git', 'grep', '-lE', pat, rev, '--', *SCOPE], capture_output=True, text=True).stdout
    return [f.split(':', 1)[1] for f in out.split()]


def brace_end(lines, a):
    d, seen = 0, False
    for i in range(a, len(lines)):
        d += lines[i].count('{') - lines[i].count('}')
        seen = seen or '{' in lines[i]
        if seen and d == 0:
            return i


def block_end(lines, a):
    for i in range(a, len(lines)):
        if '{' in lines[i]:
            return brace_end(lines, a)
        if lines[i].rstrip().endswith(';'):
            return None


def statement_end(lines, a):
    j = a
    while not lines[j].rstrip().endswith(';'):
        j += 1
    return j


def marker_ranges(lines, markers, statements, nested=True):
    ranges = []
    for i, l in enumerate(lines):
        if any(m in l and re.fullmatch(r'\s*([\w:<>&*,]+\s+)*', l[:l.index(m)]) for m in markers):
            e = block_end(lines, i)
            if e is not None and (nested or not any(a <= i <= b for a, b in ranges)):
                ranges.append((i, e))
        if any(m in l for m in statements):
            ranges.append((i, statement_end(lines, i)))
    return ranges


def dw_ranges(f, lines):
    found = []
    start = lambda marker: [i for i, l in enumerate(lines) if marker in l]
    if f.endswith('Runtime.PointNeighborhoodConfigCodecs.cpp'):
        ra = start('kDensityWeightFields[]')[0]
        found.append((ra, brace_end(lines, ra)))
        ca = start('DensityWeightConfig ParseDensityWeightConfig(')[0]
        na = max(i for i in range(ca) if lines[i].startswith('namespace Extrinsic::Runtime'))
        found.append((na, brace_end(lines, na)))
    if f.endswith('Sandbox.MeshProcessingPanels.cpp'):
        wa = start('void MeshProcessingPanels::Impl::DrawDensityWeightsWindow(')[0]
        found.append((wa, brace_end(lines, wa)))
        ga = start('RegisterWindow("view.density_weights"')[0]
        found.append((ga, statement_end(lines, ga)))
    for marker in ('struct EditorDensityWeightResult',
                   'RuntimeEngineConfigApplyResult ApplyEditorDensityWeightConfig(',
                   'std::optional<DensityWeightConfig> GetEditorDensityWeightConfig(const EditorProcessingCommands& commands)'):
        for a in start(marker):
            if not lines[a].rstrip().endswith(';'):
                found.append((a, brace_end(lines, a)))
    return found


def pilot(name, cfg, method, window, short):
    entry = [f'UpdateEditorPointScalarResult(Editor{cfg}Result', f'StartEditor{cfg}Transaction(',
             f'MakeEditor{cfg}TransactionForTest(', f'PreviewEditor{cfg}Command(', f'GetEditor{cfg}InputCatalog(',
             f'ApplyEditor{cfg}Command(', f'ApplyEditorConfigured{cfg}(', f'ApplyEditor{cfg}Config(',
             f'GetEditor{cfg}Config(']
    codec = [f'k{cfg}Fields[]{{', f'{cfg}Config Parse{cfg}Config(', f'ToString({cfg}Backend',
             f'std::string Serialize{cfg}Config(', f'Validate{cfg}ConfigSection(', f'Get{cfg}Config(',
             f'void Set{cfg}Config(', f'{cfg}ConfigFieldSpecs() noexcept', f'Make{cfg}ConfigSectionRegistration()']
    markers = tuple(entry + codec + [f'struct {method}', f'struct Editor{cfg}Result',
                                     f'void MeshProcessingPanels::Impl::{window}('])
    return dict(pat=rf'{cfg}|{name}|{window}|{short}State|{short}Transaction|{short.lower()}_progress|\b{short}\.',
                own=lambda f: f'/Modules/{cfg}/' in f, skip=lambda f: 'Runtime.PointCloudConsolidationGpu.cpp' in f,
                ranges=lambda f, lines: marker_ranges(lines, markers, (f'RegisterWindow("view.{name}"',), False))


OUTLIER_MARKERS = (
    'kOutlierAnalysisFields[]{', 'OutlierAnalysisConfig ParseOutlierAnalysisConfig(',
    'const char* ToString(OutlierAnalysisMethod', 'const char* ToString(OutlierAnalysisBackend',
    'const char* ToString(OutlierAnalysisOperation', 'std::string SerializeOutlierAnalysisConfig(',
    'ValidateOutlierAnalysisConfigSection(', 'std::optional<OutlierAnalysisConfig> GetOutlierAnalysisConfig(',
    'void SetOutlierAnalysisConfig(', 'OutlierAnalysisConfigFieldSpecs() noexcept',
    'MakeOutlierAnalysisConfigSectionRegistration()', 'struct EditorOutlierAnalysisResult',
    'struct EditorOutlierTransactionSnapshot', 'RuntimeEngineConfigApplyResult ApplyEditorOutlierAnalysisConfig(',
    'void MeshProcessingPanels::Impl::DrawOutliersWindow(')
OPS = {
    'kernel_density': pilot('kernel_density', 'KernelDensity', 'KernelDensityMethod', 'DrawDensityWindow', 'Density'),
    'point_spacing': pilot('point_spacing', 'PointSpacing', 'PointSpacingMethod', 'DrawSpacingWindow', 'Spacing'),
    'density_weight': dict(
        pat=r'DensityWeight|density_weight|DrawDensityWeightsWindow|WeightsState|WeightTransaction|weights_progress',
        own=lambda f: '/Modules/DensityWeight/' in f or f.endswith('PointAnalysisOperations.DensityWeights.cpp'),
        skip=lambda f: 'PointCloudConsolidation' in f, ranges=dw_ranges),
    'outlier': dict(
        pat=r'OutlierAnalysis|outlier_analysis|OutlierTransaction|DrawOutliersWindow|OutliersState|\bOutliers\b|'
            r'outlier_progress|AnalysisStamp',
        own=lambda f: '/Modules/OutlierAnalysis/' in f or 'PointAnalysisOperations.Outliers.cpp' in f,
        skip=lambda f: False,
        ranges=lambda f, lines: marker_ranges(lines, OUTLIER_MARKERS, ('RegisterWindow("view.outlier_analysis"',))),
}


def claims(rev):
    result, script_total = {}, {}
    for name, c in OPS.items():
        per, total = {}, 0
        for f in files(rev, c['pat']):
            if f.startswith(('src/geometry/', 'src/graphics/')) or (f.endswith('.md') and 'generated' not in f) \
                    or c['skip'](f):
                continue
            text = read(rev, f)
            lines = text.split('\n')
            if c['own'](f):
                s = set(range(len(lines) - (1 if text.endswith('\n') else 0)))
                total += len(s)
            else:
                ranges = c['ranges'](f, lines)
                inside = lambda i: any(a <= i <= b for a, b in ranges)
                named = {i for i, l in enumerate(lines) if not inside(i) and re.search(c['pat'], l)}
                s = named.union(*(set(range(a, b + 1)) for a, b in ranges))
                total += len(named) + sum(b - a + 1 for a, b in ranges)
            per[f] = s
        result[name], script_total[name] = per, total
    return result, script_total


def deoverlapped(rev):
    cl, script_total = claims(rev)
    owners = defaultdict(lambda: defaultdict(int))
    for name, per in cl.items():
        for f, s in per.items():
            for i in s:
                owners[f][i] += 1
    own = {name: {f: {i for i in s if owners[f][i] == 1} for f, s in per.items()} for name, per in cl.items()}
    shared = {f: {i for i, k in o.items() if k > 1} for f, o in owners.items()}
    return own, {f: s for f, s in shared.items() if s}, cl, script_total
```

```python
# framework_reconcile.py (imports footprint_lines)
import re, subprocess
from collections import defaultdict
from footprint_lines import deoverlapped, SCOPE
BASE = 'f96a13e2d'
PROGRAM = re.compile(r'^(RUNTIME-34[6-9]|RUNTIME-35[1-4]):')


def git(*a):
    return subprocess.run(['git', *a], capture_output=True, text=True, check=True).stdout


def per_file(rev):
    counts = defaultdict(int)
    for per in deoverlapped(rev)[0].values():
        for f, s in per.items():
            counts[f] += len(s)
    return counts


def reconcile(end):
    commits = [l.split(' ', 1) for l in git('log', '--format=%h %s', f'{BASE}..{end}').splitlines()]
    program = [h for h, s in commits if PROGRAM.match(s)]
    touched = set()
    for h in program:
        touched |= set(git('diff', '--no-renames', '--name-only', f'{h}^', h, '--', *SCOPE).split())
    net = {}
    for l in git('diff', '--no-renames', '--numstat', BASE, end, '--', *SCOPE).splitlines():
        a, d, f = l.split('\t')
        if f in touched and a != '-':
            net[f] = int(a) - int(d)
    before, after = per_file(BASE), per_file(end)
    rows = [(net[f] - (after.get(f, 0) - before.get(f, 0)), net[f], after.get(f, 0) - before.get(f, 0), f)
            for f in sorted(net)]
    others = defaultdict(list)
    for h, s in commits:
        if h not in program:
            for f in set(git('diff', '--no-renames', '--name-only', f'{h}^', h, '--', *SCOPE).split()) & set(net):
                others[f'{h} {s[:70]}'].append(f)
    return program, rows, others   # shared growth = sum of the first column
```

```python
# phase2_projection.py (imports both)
import subprocess
from footprint_lines import deoverlapped
from framework_reconcile import reconcile
GATE = 5000


def totals(rev):
    return {name: sum(len(s) for s in per.values()) for name, per in deoverlapped(rev)[0].items()}


before, after = totals('f96a13e2d'), totals('69ced3262')
END = '0348dbbed'
shared_growth = sum(r for r, *_ in reconcile(END)[1])
table = subprocess.run(['git', 'show', f'{END}:docs/architecture/declared-operations-table.md'],
                       capture_output=True, text=True, check=True).stdout
rows = []
for l in table.splitlines():
    if l.startswith('| ') and not l.startswith('| Operation'):
        c = [x.strip() for x in l.strip().strip('|').split('|')]
        inout, now = c[2], int(c[10])
        src, gpu = inout.split('→')[0].strip(), inout.rsplit(';', 1)[1].strip()
        group = 'S1' if src == 'PS' and gpu in ('sc', '–') else 'S2' if c[0] == 'outlier' else 'U'
        rows.append((c[0], group, now, 'Cu' in inout))
saving = {k: before[k] - after[k] for k in before}
measured = sum(saving.values())
s1 = [k for k in before if k != 'outlier']
rate_s1 = sum(saving[k] for k in s1) / sum(before[k] for k in s1)
rate_out = saving['outlier'] / before['outlier']
rate_dw = saving['density_weight'] / before['density_weight']
rate_all = measured / sum(before.values())
s1_rest = sum(now for n, g, now, _ in rows if g == 'S1' and n not in before)
u_lines = sum(now for n, g, now, _ in rows if g == 'U')
u_custom = sum(now for n, g, now, cu in rows if g == 'U' and cu)
base = measured + s1_rest * rate_s1 - shared_growth
for name, u in {'C': 0.0, 'B': u_lines * rate_all,
                'A': u_custom * rate_out + (u_lines - u_custom) * rate_dw}.items():
    print(name, round(base + u), '+-', round(0.1 * (s1_rest * rate_s1 + u)))
print('break-even U rate', (GATE - base) / u_lines)
```
