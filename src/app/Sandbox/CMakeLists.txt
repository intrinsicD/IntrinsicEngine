if(INTRINSIC_BUILD_SANDBOX AND NOT INTRINSIC_HEADLESS_NO_GLFW)
    add_executable(ExtrinsicSandbox)
    set_target_properties(ExtrinsicSandbox PROPERTIES CXX_SCAN_FOR_MODULES ON)
    target_compile_options(ExtrinsicSandbox PRIVATE ${INTRINSIC_COMPILE_FLAGS})
    target_compile_features(ExtrinsicSandbox PUBLIC cxx_std_23)
    target_link_options(ExtrinsicSandbox PRIVATE ${INTRINSIC_LINK_FLAGS})

    target_sources(ExtrinsicSandbox
        PRIVATE
        main.cpp
    )

    target_link_libraries(ExtrinsicSandbox
        PRIVATE
        ExtrinsicRuntime
        ExtrinsicSandboxEditor
        nlohmann_json::nlohmann_json
    )
    # ARCH-022: the feature plugins, each of which adds itself to INTRINSIC_STARTUP_PLUGINS (so the Sandbox names none,
    # and a new plugin is a new folder). Shared builds load (and reload) their MODULEs beside the executable; static
    # builds link their objects and register them in-process. SandboxStartupPlugins.hpp carries the list into main.cpp.
    get_property(startup_plugins GLOBAL PROPERTY INTRINSIC_STARTUP_PLUGINS)
    set(startup_imports "")
    set(startup_entries "")
    set(startup_files "")
    foreach(plugin IN LISTS startup_plugins)
        string(APPEND startup_imports "import Intrinsic.Plugin.${plugin};\n")
        string(APPEND startup_entries "{&Intrinsic::Plugins::${plugin}::Register, &Intrinsic::Plugins::${plugin}::Shutdown}, ")
        string(APPEND startup_files "\"libIntrinsic${plugin}Plugin.so\", ")
        if(BUILD_SHARED_LIBS)
            add_dependencies(ExtrinsicSandbox Intrinsic${plugin}Plugin)
        else()
            target_link_libraries(ExtrinsicSandbox PRIVATE Intrinsic${plugin}PluginObjects)
        endif()
    endforeach()
    if(BUILD_SHARED_LIBS)
        set(startup_imports "")
    else()
        target_compile_definitions(ExtrinsicSandbox PRIVATE INTRINSIC_PLUGINS_IN_PROCESS)
    endif()
    file(CONFIGURE OUTPUT ${CMAKE_CURRENT_BINARY_DIR}/generated/SandboxStartupPlugins.hpp CONTENT
"// Generated by src/app/Sandbox/CMakeLists.txt from INTRINSIC_STARTUP_PLUGINS.
#pragma once
${startup_imports}#define INTRINSIC_STARTUP_PLUGIN_ENTRIES ${startup_entries}
#define INTRINSIC_STARTUP_PLUGIN_FILES ${startup_files}
")
    target_include_directories(ExtrinsicSandbox PRIVATE ${CMAKE_CURRENT_BINARY_DIR}/generated)

    include(${CMAKE_SOURCE_DIR}/cmake/CompileShaders.cmake)

    # Sandbox resolves promoted runtime shaders beside its executable.
    intrinsic_add_glsl_shaders(ExtrinsicSandbox)
endif()
